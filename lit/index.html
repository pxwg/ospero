Ospero文学编程笔记

从 Source 到消费者

同一处引用，有三种读法。作者读它的理由，编辑器要知道点哪里能跳到定义，tangler 要知道该把哪段代码插进文件。Ospero 把三者共有的事实留在一次观察里，各个消费者再取自己需要的部分。

沿一行 <<sum-step>> 走一遍。它出现在 Org 的求和示例里：作者先讲求和，再讲怎样累加；生成的 OCaml 文件却需要先定义累加函数。点正文中的链接，可以把理由放在右侧对读；展开段末的卡片，可以留在原处看实现。

从一段 Org 认出一张卡

刚读入文件时，适配器拿到的是文档、语言和原文。Org 扫描器认出标题、属性抽屉和源块。卡片层用标题的 ID 划定范围，把标题、元数据、正文和它们在文件里的位置交给前端。一个普通小标题仍属于父卡；一段示例代码里的标题符号则只是代码。换用 Markdown 或 Typst，只需把这门语言的文档交给相应的适配器。

此时的 Source.note 已经有形状。正文是一列文本与动作。读到具名源块，playground 的规则把它记成一次 code-block 调用；块内的 <<sum-step>> 仍随整段代码保存。因而扫描器可以完整保留示例里的原文，把拼接的含义留给懂代码块的人。

module Source : sig
  type 'action content =
    | Text of Text.t
    | Action of 'action

  (** Constructing a note does not register its identity.
      Metadata follows the project schema; the body preserves input order.
      Each body occurrence carries its optional source span independently
      of the text or action value. The outer body wrapper locates the whole
      container, including an explicitly empty body. *)
  type ('metadata, 'action) note = {
    id : Graph.Vertex_id.t Provenance.located;
        (** The span covers the ID as written, such as a label or property. *)
    title : Text.t Provenance.located;
    metadata : 'metadata Provenance.located;
        (** The optional span covers the original metadata as a whole,
            independently of the project-defined decoded value.
            Omitted metadata has no direct source span. *)
    body : 'action content Provenance.located list Provenance.located;
  }
end

(** Registered identities are frozen as a set before observation.
    Initial notes need not cover every registered identity.
    Duplicate registrations are harmless; duplicate definitions are conflicts. *)
type ('metadata, 'action) project = {
  identities : Graph.Vertex_id.t list;
  notes : ('metadata, 'action) Source.note Provenance.located list;
}

(** The immediate action that generated a definition. Its owning note can
    itself be generated; this link does not replace direct source locations. *)
type generation = {
  source : Graph.Vertex_id.t;
  action : Provenance.span option;
}

module Observed : sig
  (** Note fields retain direct source locations until assembly separates
      semantic values from the observation's source registry. *)
  type 'metadata note = {
    id_span : Provenance.span option;
    title : Text.t Provenance.located;
    metadata : 'metadata Provenance.located;
    generated_from : generation option;
  }

  (** Final body data, with no pending observation actions.
      Consumption declarations are interpreted by consumers; they do not
      contribute further definitions or graph edges. *)
  type 'consumption content =
    | Text of Text.t
    | Consume of 'consumption
end

(** Ordered contributions returned by an action interpreter. *)
type ('metadata, 'relation, 'action, 'consumption, 'error) contribution =
  | Define of ('metadata, 'action) Source.note Provenance.located
      (** Contribute a definition for a registered identity.
          The core observes its body under that identity; subsequent
          contributions still belong to the original current note.
          Duplicate definitions are conflicts. The wrapper records the direct
          definition span, not the generating action's span. *)
  | Relate of {
      target : Graph.Vertex_id.t Provenance.located;
      relation : 'relation;
    }
      (** Contribute an edge from the current note to the target. The span
          locates the target argument within the action. Each occurrence
          receives an ID from its source and local index. Contributions of
          one action that name the same target at the same span form one
          source reference, as when an argument yields several relations. *)
  | Emit of 'consumption Observed.content
      (** Append action-free content to the current note's observed body. *)

  | Fail of 'error
      (** Report a failed action contribution without discarding siblings. *)

(** Interpret an action into ordered contributions.
    The action carries its explicit inputs; the core supplies no context.
    Contributions belong to the note whose body contains the action. *)
type ('metadata, 'relation, 'action, 'consumption, 'error) interpreter =
  'action ->
  ('metadata, 'relation, 'action, 'consumption, 'error) contribution list

源块声明了语言、片段 ID，以及可选的输出路径。适配器先交出带位置的字段；前端按动作声明的规格，把字段解码成项目能解释的值。同样的数字串，写在 ID 栏要保留拼法，写在整数栏才做数值转换。参数声明约定了这次转换，同一份 schema 也能告诉补全器可以填写哪些字段。

module Payload = struct
  type t = value Provenance.located

  and value =
    | String of string
    | Bool of bool
    | Int of int
    | List of t list
    | Record of field list
    | Contents of Text.t

  and field = {
    name : string Provenance.located;
    value : t;
  }

  module Kind_type = struct
    type t = String | Bool | Int | List | Record | Contents
  end

  let kind_of : value -> Kind_type.t = function
    | String _ -> Kind_type.String
    | Bool _ -> Kind_type.Bool
    | Int _ -> Kind_type.Int
    | List _ -> Kind_type.List
    | Record _ -> Kind_type.Record
    | Contents _ -> Kind_type.Contents

  module Kind = struct
    include Kind_type

    let of_value = kind_of
  end
end

module Invocation = struct
  type t = {
    name : string Provenance.located;
    arguments : Payload.field list;
  }
end

module Diagnostic = struct
  type segment =
    | Field of string
    | Index of int

  type problem =
    | Type_mismatch of { expected : Payload.Kind.t; given : Payload.Kind.t }
    | Missing_field of string
    | Unknown_field of { name : string; known : string list }
    | Duplicate_field of string
    | Invalid_choice of { given : string; choices : string list }
    | Unknown_action of { name : string; known : string list }
    | Invalid of string
    | Missing_identity of { insertion : Provenance.span option; prefix : string; suffix : string }

  type t = {
    problem : problem;
    path : segment list;
    span : Provenance.span option;
  }

  let make ?(path = []) ?span problem = { problem; path; span }

  let with_fallback_span span diagnostic =
    match diagnostic.span, span with
    | None, Some _ -> { diagnostic with span }
    | _ -> diagnostic
end

module type DECODING = sig
  type metadata
  type action

  val decode_metadata :
    Payload.t -> (metadata, Diagnostic.t list) result

  val decode_action :
    Invocation.t -> (action, Diagnostic.t list) result
end

type 'a read = ('a, Diagnostic.t list) result
type raw_note = (Payload.value read, Invocation.t read) Observation.Source.note
type raw_project = (Payload.value read, Invocation.t read) Observation.project

type 'a decoded = ('a, Diagnostic.t list) result

type action_failure =
  | Unread of Diagnostic.t list
  | Undecoded of Invocation.t * Diagnostic.t list

type 'a decoded_action = ('a, action_failure) result

type 'error failure =
  | Metadata_decode of Diagnostic.t list
  | Action_decode of action_failure
  | Interpretation of 'error

module type DECODER = sig
  type metadata
  type action

  val decode_source :
    raw_note Provenance.located ->
    (metadata decoded, action decoded_action) Observation.Source.note Provenance.located

  val decode_project :
    raw_project -> (metadata decoded, action decoded_action) Observation.project
end

module Decode (Rules : DECODING) = struct
  type metadata = Rules.metadata
  type action = Rules.action

  let decode_content (content : _ Provenance.located) =
    match content.value with
    | Observation.Source.Text text ->
        { content with value = Observation.Source.Text text }
    | Observation.Source.Action read ->
        let locate = List.map (Diagnostic.with_fallback_span content.span) in
        let decoded = match read with
          | Error diagnostics -> Error (Unread (locate diagnostics))
          | Ok invocation ->
              Rules.decode_action invocation
              |> Result.map_error (fun diagnostics -> Undecoded (invocation, locate diagnostics))
        in
        { content with value = Observation.Source.Action decoded }

  let decode_source (located : _ Observation.Source.note Provenance.located) =
    let note = located.value in
    let metadata_span =
      match note.metadata.span with
      | Some _ as span -> span
      | None -> note.title.span
    in
    let metadata =
      Result.bind note.metadata.value (fun value ->
        Rules.decode_metadata { note.metadata with value })
      |> Result.map_error (List.map (Diagnostic.with_fallback_span metadata_span))
    in
    { located with value =
      { Observation.Source.id = note.id;
        title = note.title;
        metadata = { note.metadata with value = metadata };
        body = { note.body with value = List.map decode_content note.body.value } } }

  let decode_project (project : (_, _) Observation.project) =
    { project with notes = List.map decode_source project.notes }
end

module type FRONTEND = sig
  include DECODER

  type relation
  type consumption
  type error

  val interpret :
    action ->
    (metadata decoded, relation, action decoded_action, consumption, error)
      Observation.contribution list
end

module Make (Frontend : FRONTEND) = struct
  let observe project =
    let interpret = function
      | Error failure -> [Observation.Fail (Action_decode failure)]
      | Ok action -> List.map (function
          | Observation.Define note -> Observation.Define note
          | Observation.Relate { target; relation } -> Observation.Relate { target; relation }
          | Observation.Emit content -> Observation.Emit content
          | Observation.Fail error -> Observation.Fail (Interpretation error))
          (Frontend.interpret action)
    in
    let metadata_errors = function
      | Ok _ -> []
      | Error diagnostics -> [Metadata_decode diagnostics]
    in
    let declarations, issues =
      Observation.expand ~metadata_errors ~interpret (Frontend.decode_project project)
    in
    Observation.assemble ~issues declarations
end

let expand ?(metadata_errors = fun _ -> []) ~interpret project =
  let identities = Vertex_set.of_list project.identities in
  let definitions = ref Vertex_set.empty in
  let ambiguous = ref Vertex_set.empty in
  let pending = Queue.create () in
  let issues = ref [] in
  let report span issue = issues := Provenance.located ?span issue :: !issues in
  let vertices = ref [] in
  let edges = ref [] in
  let reference_count = ref 0 in
  let bodies = ref [] in
  let register generated_from (located : (_, _) Source.note Provenance.located) =
    let note = located.value in
    List.iter
      (fun error -> report note.metadata.span (Metadata_error (note.id.value, error)))
      (metadata_errors note.metadata.value);
    let id = note.id.value in
    if not (Vertex_set.mem id identities) then
      report (at_id note.id.span located.span) (Unregistered_definition id)
    else begin
      let vertex =
        { Graph.id;
          note = { Observed.id_span = note.id.span; title = note.title;
                   metadata = note.metadata; generated_from } }
      in
      vertices := { located with value = vertex } :: !vertices;
      if Vertex_set.mem id !definitions then
        ambiguous := Vertex_set.add id !ambiguous
      else begin
        definitions := Vertex_set.add id !definitions;
        Queue.add note pending
      end
    end
  in
  List.iter (register None) project.notes;
  let observe_body (note : (_, _) Source.note) =
    let source = note.id.value in
    let contents = ref [] in
    let edge_index = ref 0 in
    let emit span content =
      contents := Provenance.located ?span content :: !contents
    in
    (* [arguments] holds the references already opened by the current action. *)
    let contribute span arguments = function
      | Define definition ->
          register (Some { source; action = span }) definition
      | Relate { target; relation } ->
          let id = Observation_edge_id.make ~source ~index:!edge_index in
          incr edge_index;
          let key = target.value, target.span in
          let reference = match List.assoc_opt key !arguments with
            | Some reference -> reference
            | None ->
                let reference = !reference_count in
                incr reference_count;
                arguments := (key, reference) :: !arguments;
                reference
          in
          edges := { edge = { Graph.id; source; target = target.value; relation };
                     reference; call = span; target_span = target.span } :: !edges
      | Emit content -> emit span content
      | Fail error -> report span (Action_error (source, error))
    in
    List.iter
      (fun (content : _ Provenance.located) ->
        match content.value with
        | Source.Text text -> emit content.span (Observed.Text text)
        | Source.Action action ->
            List.iter (contribute content.span (ref [])) (interpret action))
      note.body.value;
    bodies := (source, { note.body with value = List.rev !contents }) :: !bodies
  in
  (* Initial notes run first; generated definitions join the FIFO queue.
     Assembly also rejects contributions from definitions found ambiguous later. *)
  while not (Queue.is_empty pending) do
    let note = Queue.take pending in
    if not (Vertex_set.mem note.id.value !ambiguous) then observe_body note
  done;
  { vertices = List.rev !vertices; edges = List.rev !edges; bodies = List.rev !bodies },
  List.rev !issues

          Some (located.value.Graph.id :: Option.value children ~default:[])) index)
    Vertex_map.empty declarations.vertices

    | [] -> rejected

        reject (Vertex_set.add id rejected) (List.rev_append children rest)

        [ record span (Duplicate_definition vertex.id) ]
      else match note.generated_from with
        | Some { source; _ } when not (available source) ->
            [ record span (Unavailable_generator (vertex.id, source)) ]
        | _ -> []
    in
    located,

    Edge_map.update declared.edge.Graph.id

      else [ reference_error declared.call edge (Graph.Duplicate_edge edge.id) ]

      | None ->

                      (Graph.Missing_endpoint (edge.id, Source, edge.source)) ])

          Hashtbl.add endpoint_failures declared.reference failures;

    | [] -> Either.Left located

    let groups = List.fold_left (fun groups ((declared : _ edge_declaration), reasons) ->

        let first, edges, rejected = Option.value group ~default:(declared, [], []) in
        Some (first, declared.edge :: edges, List.rev_append reasons rejected))
        groups) Int_map.empty checked
    in
    let once records = List.fold_left (fun kept record ->
      if List.exists (fun r -> Failure.compare r.Failure.id record.Failure.id = 0) kept then kept
      else record :: kept) [] records |> List.rev in
    (* Ordinals follow the first edge of each reference, hence declaration order. *)
    Int_map.bindings groups |> List.map (fun (_, ((first : _ edge_declaration), edges, rejected)) ->
      { source = first.edge.Graph.source;
        target = Provenance.located ?span:first.target_span first.edge.target;
        call = first.call; edges = List.rev edges; rejected = once (List.rev rejected) })
  in
  let body_index = List.fold_left (fun index (id, (contents : _ Provenance.located)) ->
    if available id then
      Vertex_map.update id (function
        | None -> Some [contents]
        | Some candidates -> Some (contents :: candidates)) index

      if not (Vertex_map.mem id vertex_index) then report contents.span (Unknown_body_vertex id);
      index

    let id = located.value.Graph.id in
    match Vertex_map.find_opt id body_index with
    | Some [contents] -> Vertex_map.add id contents bodies
    | Some candidates ->
        List.iter (fun (body : _ Provenance.located) -> report body.span (Duplicate_body id))
          (List.rev candidates);
        bodies
    | None -> report located.span (Missing_body id); bodies) Vertex_map.empty vertices
  in

    let values = List.map (fun (located : _ Provenance.located) ->
      let vertex : _ Observed.note Graph.vertex = located.value in
      { vertex with note = { Graph.title = vertex.note.title.value;
                             metadata = vertex.note.metadata.value } }) vertices in

    | Ok graph -> graph
    | Error _ -> invalid_arg "Observation.assemble: invalid resolved topology"
  in

    Vertex_map.add located.value.Graph.id (f located) index) Vertex_map.empty vertices in
  let edge_sources = List.fold_left (fun index (located : _ Provenance.located) ->
    Edge_map.add located.value.Graph.id located.span index) Edge_map.empty edges in

    Vertex_map.update (issue_owner failure.Failure.detail.Provenance.value)
      (fun group -> Some (failure :: Option.value group ~default:[])) index)
    failures Vertex_map.empty in
  let rejected_by_source = List.fold_right (fun reference index ->
    Vertex_map.update reference.declaration.value.Graph.source
      (fun group -> Some (reference :: Option.value group ~default:[])) index)
    rejected_references Vertex_map.empty in

    Vertex_map.update (key item) (fun group -> Some (item :: Option.value group ~default:[])) index)
    items Vertex_map.empty in

"""Tangle and weave the Org subset used by Ospero's bootstrap cards."""
import argparse
from dataclasses import dataclass
from html import escape
from pathlib import Path
import re
import shlex
import sys
from tempfile import NamedTemporaryFile


@dataclass
class Block:
    id: str
    language: str
    target: str
    noweb: bool
    body: str


LINK = re.compile(r"\[\[id:([^\]]+)\](?:\[([^\]]*)\])?\]")
INCLUDE = re.compile(r"<<([^\s<>]+)>>")


def load(workspace):
    notes, blocks, titles = [], {}, {}
    for path in sorted((workspace / "notes").rglob("*.org")):
        lines = iter(path.read_text().splitlines(keepends=True))
        parts = None
        for line in lines:
            if parts is None and line.strip() and not line.startswith("* "):
                raise ValueError(f"{path}: expected a card heading")
            if re.match(r"^#\+begin_src\s", line, re.I):
                tokens = shlex.split(line.strip())[1:]
                if not tokens or tokens[0].startswith(":"):
                    raise ValueError(f"{path}: expected a source language")
                language, *args = tokens
                headers = dict(zip(args[::2], args[1::2]))
                if len(headers) * 2 != len(args):
                    raise ValueError(f"{path}: malformed block header")
                body = []
                for raw in lines:
                    if raw.strip().lower() == "#+end_src":
                        break
                    body.append(raw)
                else:
                    raise ValueError(f"{path}: unterminated source block")
                if ":id" in headers:
                    flag = headers.get(":zk-noweb", "no")
                    if flag not in ("yes", "no"):
                        raise ValueError(f"{path}: invalid :zk-noweb")
                    block = Block(headers[":id"], language, headers.get(":target-file", ""),
                                  flag == "yes", "".join(body))
                    if block.id in blocks:
                        raise ValueError(f"{path}: duplicate block {block.id}")
                    blocks[block.id] = block
                    parts.append(block)
                else:
                    if ":target-file" in headers or ":zk-noweb" in headers:
                        raise ValueError(f"{path}: code block requires :id")
                    parts.append(Block("", language, "", False, "".join(body)))
            elif line.startswith("* "):
                title = line[2:].strip()
                if next(lines, "").strip() != ":PROPERTIES:":
                    raise ValueError(f"{path}: card needs a property drawer")
                key = next(lines, "").strip().split(":ID:", 1)
                if len(key) != 2 or key[0] or next(lines, "").strip() != ":END:":
                    raise ValueError(f"{path}: expected :ID: and :END:")
                id_ = key[1].strip()
                if not id_ or id_ in titles:
                    raise ValueError(f"{path}: invalid or duplicate card {id_}")
                titles[id_] = title
                parts = []
                notes.append((id_, title, parts))
            elif line.strip():
                parts.append(line.rstrip())
    if not notes:
        raise ValueError("no cards in notes/")
    if titles.keys() & blocks.keys():
        raise ValueError("card and block share an ID")
    for _, _, parts in notes:
        for part in parts:
            if isinstance(part, str):
                for target, _ in LINK.findall(part):
                    if target not in titles and target not in blocks:
                        raise ValueError(f"unknown link {target}")
    return notes, blocks


def included(block, line):
    value = line.strip()
    if block.noweb and value.startswith("<<") and value.endswith(">>"):
        match = INCLUDE.fullmatch(value)
        if not match:
            raise ValueError(f"{block.id}: malformed include {value}")
        return match
    return None


def compile_blocks(blocks):
    expanded, outputs = {}, {}

    def expand(block, stack):
        if block.id in stack:
            raise ValueError(" -> ".join((*stack, block.id)))
        if block.id not in expanded:
            chunks = []
            for line in block.body.splitlines(keepends=True):
                ref = included(block, line)
                if ref:
                    child = blocks.get(ref[1])
                    if child is None or child.language != block.language:
                        raise ValueError(f"{block.id}: incompatible or missing block {ref[1]}")
                    line = expand(child, (*stack, block.id))
                chunks.append(line)
            expanded[block.id] = "".join(chunks)
        return expanded[block.id]

    for block in blocks.values():
        body = expand(block, ())
        if block.target:
            path = Path(block.target)
            if path.is_absolute() or ".." in path.parts or not path.name or block.target.endswith("/") or path in outputs:
                raise ValueError(f"invalid or duplicate target {block.target}")
            outputs[path] = body
    for path in outputs:
        if any(parent in outputs for parent in path.parents):
            raise ValueError(f"conflicting target {path}")
    return outputs


def tangle(workspace, outputs):
    out = workspace / "out"
    for path, body in outputs.items():
        target = out / path
        if any(p.is_symlink() for p in [target, *target.parents] if p == out or out in p.parents):
            raise ValueError(f"symlink in output path {target}")
        target.parent.mkdir(parents=True, exist_ok=True)
        with NamedTemporaryFile(dir=target.parent, delete=False) as staged:
            try:
                staged.write(body.encode())
                staged.close()
                Path(staged.name).replace(target)
            finally:
                Path(staged.name).unlink(missing_ok=True)


def link(target, label):
    return f'<a href="#{escape(target, quote=True)}">{escape(label)}</a>'


def prose(line, titles):
    pieces, start = [], 0
    for match in LINK.finditer(line):
        target, alias = match.groups()
        label = alias if alias is not None else titles.get(target, target)
        pieces.extend((escape(line[start:match.start()]), link(target, label)))
        start = match.end()
    return "".join(pieces) + escape(line[start:])


def weave(notes):
    titles = {key: title for key, title, _ in notes}
    print('<!doctype html><html lang="zh"><meta charset="utf-8"><title>Ospero · 文学卡片</title>')
    print('<style>body{max-width:56rem;margin:3rem auto;padding:0 1rem;line-height:1.7}'
          'pre{overflow:auto;background:#f5f4f1;padding:1rem}summary{cursor:pointer}</style>')
    print('<h1>Ospero · 文学卡片</h1><nav>')
    for id_, title, _ in notes:
        print('<div>' + link(id_, title) + '</div>')
    print('</nav>')
    for id_, title, parts in notes:
        print(f'<article id="{escape(id_)}"><h2>{escape(title)}</h2>')
        for part in parts:
            if isinstance(part, str):
                if part.startswith("** "):
                    print('<h3>' + prose(part[3:], titles) + '</h3>')
                else:
                    print('<p>' + prose(part, titles) + '</p>')
                continue
            print(f'<details id="{escape(part.id)}" data-target="{escape(part.target)}"><summary>'
                  f'{escape(part.target or part.id or part.language)}</summary><pre><code>', end="")
            for line in part.body.splitlines(keepends=True):
                ref = included(part, line)
                if ref:
                    print(line.replace(ref[0], link(ref[1], ref[0])), end="")
                else:
                    print(escape(line), end="")
            print('</code></pre></details>')
        print('</article>')
    print('</html>')


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=["tangle", "weave"])
    parser.add_argument("workspace", type=Path, nargs="?", default=Path(__file__).resolve().parent)
    args = parser.parse_args()
    notes, blocks = load(args.workspace)
    outputs = compile_blocks(blocks)
    if args.command == "tangle":
        tangle(args.workspace, outputs)
    else:
        weave(notes)


if __name__ == "__main__":
    try:
        main()
    except (ValueError, OSError) as error:
        sys.exit(str(error))
