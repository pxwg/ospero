(** The default Org adapter.

    A heading whose property drawer has an [:ID:] declares a card; its other
    properties are metadata, keys read in lower case, a dotted key such as
    [:CAPTURE.TYPE:] naming a field of a record field. A title omits a leading
    task keyword (those of the file's [#+TODO:] sequences, else TODO and DONE),
    a priority cookie and trailing tags. A COMMENT heading and its subtree are
    not read. A named call is
    [#+zk_review: :target b] without a body, or a [#+begin_zk_review] block
    whose contents are the body. Arguments are written [:key value]; a value
    is read by the declared field type, so [:target 2604032336] is a string in
    a string field. Lists are written [\[a, "b c"\]] or as space-separated
    words. Lines inside a block are never headings. *)
open Ospero_kernel
open Frontend

val language : Text.Language.t

(** A value as written: the raw text after a [:key] or a property name. *)
type value = string

(** The identity rules read this at each heading. *)
type heading = {
  level : int;
  title : string Provenance.located;
  properties : (string Provenance.located * value Provenance.located) list;
}

type node =
  | Keyword of {
      name : string Provenance.located;
      value : value Provenance.located;
      inner : node Provenance.located list;  (** Links in the value. *)
    }
  | Block of {
      name : string Provenance.located;
      parameters : value Provenance.located;
      contents : Text.t Provenance.located;
      inner : node Provenance.located list;  (** Empty for verbatim blocks. *)
    }
  | Link of {
      path : string Provenance.located;  (** Such as [id:b]. *)
      description : string Provenance.located option;
    }
  | Card of Card.nested

type context = value Card.context

(** Strings, integers, booleans and lists. Records are left to project rules. *)
(** Everything in a document, for consumers that work on its text. The adapter
    reads the same outline. *)
val scan : file:string -> string -> (heading, value, node) Card.outline

(** Nodes inside a node: links in a keyword value, or nodes in a non-verbatim block. *)
val children : node -> node Provenance.located list

val values : value Rules.Payload_rules.t

(** ["id-property"]: the [:ID:] property, in any letter case. *)
val identities : heading Rules.Identity_rules.t

(** ["named-keyword"] and ["named-block"]: calls through the [zk_] prefix. *)
val actions : (node, context) Rules.Action_rules.t

(** Split [:key value] arguments. Text before the first key is an error. *)
val arguments :
  value Provenance.located ->
  (value Rules.Payload_rules.field list, Diagnostic.t list) result

(** [define] names the action whose Contents declare cards, ["define"] by
    default. Drawers often hold properties written by other tools, so those
    the metadata schema does not declare are ignored unless
    [unknown_properties] is [`Report]. *)
val adapter :
  ?values:value Rules.Payload_rules.t ->
  ?identities:heading Rules.Identity_rules.t ->
  ?actions:(node, context) Rules.Action_rules.t ->
  ?define:string ->
  ?unknown_properties:[ `Ignore | `Report ] ->
  ('metadata, 'action) Syntax.Frontend_input.t ->
  string Syntax.adapter

(** Optional sugars bound to project actions. *)
module Sugar : sig
  (** ["id-link"]: [\[\[id:b\]\]] in a body, located at the ID. *)
  val id_link :
    'args Syntax.Action.t -> (string Provenance.located -> 'args) ->
    (node, context) Rules.Action_rules.t -> (node, context) Rules.Action_rules.t

  (** ["zk-include"]: [#+zk_include: :target b]. *)
  val include_keyword :
    'args Syntax.Action.t -> (string Provenance.located -> 'args) ->
    (node, context) Rules.Action_rules.t -> (node, context) Rules.Action_rules.t

  (** ["nested-card"]: a card heading below the current card. *)
  val nested_card :
    'args Syntax.Action.t ->
    (contents:Text.t Provenance.located -> depth:int Provenance.located -> 'args) ->
    (node, context) Rules.Action_rules.t -> (node, context) Rules.Action_rules.t
end
